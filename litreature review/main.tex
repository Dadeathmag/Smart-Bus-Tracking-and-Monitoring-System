%---------------- Paper 1 ----------------%
\begin{frame}{Paper 1: Bus Monitoring System - Neha Parkara et al., IJSART 2021}

\begin{itemize}
\item \textbf{Concept:} IoT–GPS based bus monitoring and tracking
\item \textbf{Methodology:} Real time tracking using IOT-GPS based system
\item \textbf{Advantages:} Real-time tracking, improved safety, reduced waiting time
\item \textbf{Limitations and Gap identified:} Internet/GPS dependency, low security, no identification system

\end{itemize}
\end{frame}

%---------------- Paper 2 ----------------%
\begin{frame}{Paper 2: Attendance System Using Facial Recognition - Tushar Rane et al., IJCRT 2023}

\begin{itemize}
\item \textbf{Concept:} Facial recognition–based automated attendance
\item \textbf{Methodology:} Attendance is recorded automatically
\item \textbf{Advantages:} Contactless, fast, prevents proxy
\item \textbf{Limitations and Gap identified:} Lighting, angle, occlusion issues,no tracking

\end{itemize}
\end{frame}

%---------------- Paper 3 ----------------%
\begin{frame}{Paper 3: AI-Based Bus Attendance System - Johnson C et al., IJNRD 2023}

\begin{itemize}
\item \textbf{Concept:} AI-based bus attendance using facial recognition
\item \textbf{Methodology:} Attendance is marked using face recognition and barcode scanning
\item \textbf{Advantages:} Contactless, automated, prevents proxy
\item \textbf{Limitations and Gap identified:} No real time alerts

\end{itemize}
\end{frame}

%---------------- Paper 4 ----------------%
\begin{frame}{Paper 4: Real-Time Bus Tracking and Attendance - R. Aruna et al., IJSRED 2025}

\begin{itemize}
\item \textbf{Concept:} IoT-based bus tracking with RFID attendance
\item \textbf{Methodology:}Automated entry and exit are monitored
\item \textbf{Advantages:} Automated attendance, real-time tracking
\item \textbf{Limitations and Gap identified:} No computer vision or face recognition

\end{itemize}
\end{frame}

%---------------- Paper 5 ----------------%
\begin{frame}{Paper 5: Smart College Bus Tracker - M. Sowndharya et al., IJNRD 2025}

\begin{itemize}
\item \textbf{Concept:} Real-time college bus tracking with ETA prediction
\item \textbf{Methodology:} ETA is calculated dynamically
\item \textbf{Advantages:} Reduced waiting time, improved transparency
\item \textbf{Limitations and Gap identified:} No attendance

\end{itemize}
\end{frame}
